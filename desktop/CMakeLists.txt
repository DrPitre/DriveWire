cmake_minimum_required(VERSION 3.24)
project(DriveWireDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DRIVEWIRE_FETCH_WX "Build a pinned wxWidgets instead of a system installation" OFF)
set(DRIVEWIRE_SERVER "" CACHE FILEPATH "Release Swift server to bundle with the UI")
if(DRIVEWIRE_FETCH_WX)
  include(FetchContent)
  set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
  set(wxBUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
  set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(wxWidgets GIT_REPOSITORY https://github.com/wxWidgets/wxWidgets.git GIT_TAG v3.2.8 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(wxWidgets)
  set(DRIVEWIRE_WX wx::core wx::base wx::net)
else()
  find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base net)
  include(${wxWidgets_USE_FILE})
  set(DRIVEWIRE_WX ${wxWidgets_LIBRARIES})
endif()
add_executable(drivewire-desktop WIN32 MACOSX_BUNDLE src/main.cpp)
target_include_directories(drivewire-desktop PRIVATE third_party)
target_link_libraries(drivewire-desktop PRIVATE ${DRIVEWIRE_WX})
set_target_properties(drivewire-desktop PROPERTIES
  MACOSX_BUNDLE_BUNDLE_NAME "DriveWire Desktop"
  MACOSX_BUNDLE_GUI_IDENTIFIER "org.drivewire.desktop"
  MACOSX_BUNDLE_BUNDLE_VERSION "0.1.0"
  MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1.0"
  OUTPUT_NAME "DriveWire Desktop")
if(MSVC)
  target_compile_options(drivewire-desktop PRIVATE /W4 /utf-8)
else()
  target_compile_options(drivewire-desktop PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(DRIVEWIRE_SERVER)
  if(APPLE)
    set(SERVER_DEST "$<TARGET_BUNDLE_DIR:drivewire-desktop>/Contents/MacOS")
  else()
    set(SERVER_DEST "$<TARGET_FILE_DIR:drivewire-desktop>")
  endif()
  add_custom_command(TARGET drivewire-desktop POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DRIVEWIRE_SERVER}" "${SERVER_DEST}/")
endif()
install(TARGETS drivewire-desktop BUNDLE DESTINATION . RUNTIME DESTINATION .)
if(DRIVEWIRE_SERVER AND NOT APPLE)
  install(PROGRAMS "${DRIVEWIRE_SERVER}" DESTINATION .)
endif()
install(FILES README.md third_party/JSON-LICENSE.MIT DESTINATION licenses)
set(CPACK_PACKAGE_NAME DriveWire-Desktop)
set(CPACK_PACKAGE_VENDOR DriveWire)
set(CPACK_GENERATOR ZIP)
include(CPack)
